type GridProps = {
  options: string[];
  value: string | null;
  onSelect: (value: string) => void;
};

export function Grid({
  options,
  value,
  onSelect,
}: GridProps) {
  return (
    <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
      {options.map((option) => {
        const selected = value === option;

        return (
          <button
            key={option}
            type="button"
            onClick={() => onSelect(option)}
            style={{
              backgroundColor: selected ? "#ff6b35" : "#111827",
              color: "#ffffff",
              borderColor: selected ? "#ff6b35" : "#374151",
            }}
            className="min-h-[52px] px-4 py-3 rounded-2xl border text-sm font-semibold transition hover:opacity-90 shadow-sm"
          >
            {option}
          </button>
        );
      })}
    </div>
  );
}

type RateBarProps = {
  max: number;
  value: number | null;
  onSelect: (value: number) => void;
};

export function RateBar({
  max,
  value,
  onSelect,
}: RateBarProps) {
  return (
    <div
      className="grid gap-2"
      style={{
        gridTemplateColumns: `repeat(${max}, minmax(0, 1fr))`,
      }}
    >
      {Array.from({ length: max }, (_, index) => index + 1).map((number) => {
        const selected = value === number;

        return (
          <button
            key={number}
            type="button"
            onClick={() => onSelect(number)}
            style={{
              backgroundColor: selected ? "#ff6b35" : "#111827",
              color: "#ffffff",
              borderColor: selected ? "#ff6b35" : "#374151",
            }}
            className="min-h-[48px] rounded-2xl border font-semibold transition hover:opacity-90 shadow-sm"
          >
            {number}
          </button>
        );
      })}
    </div>
  );
}

      () =>

        ).map((item) => ({
          ...item,
          athlete:

              item.athleteId,
            ) ?? "Unknown athlete",
        })),
      [
        currentPeriodCheckins,
        athleteNameById,
      ],
    );

    useMemo(
      () =>
        calculateCheckinTrend(

          trendMetric,
        ),
      [
        currentPeriodCheckins,
        trendMetric,
      ],
    );

    useMemo(
      () =>
        calculateCheckinTrend(
          previousPeriodCheckins,
          trendMetric,
        ),
      [
        previousPeriodCheckins,
        trendMetric,
      ],
    );

  const trendAxisMax =
    trendMetric === "RPE"
      ? 10
      : trendMetric ===
          "Physical Feeling"

        : trendMetric ===
            "Performance Quality"
          ? 4
          : 3;

    [
      trendAxisMax,
      trendAxisMax * 0.75,
      trendAxisMax * 0.5,
      trendAxisMax * 0.25,
      0,
    ].map(
      (value) =>
        Number.isInteger(value)
          ? value
          : Number(
              value.toFixed(1),
            ),
    );

  function createTrendPoints(
    points:
      typeof trendData,
  ) {
    if (
      points.length === 0
    ) {
      return "";
    }

    return points
      .map(
        (point, index) => {
          const x =
            points.length === 1
              ? 250
              : (
                  index /
                  (
                    points.length -
                    1
                  )
                ) *
                500;

          const normalized =
            Math.max(
              0,
              Math.min(
                1,
                point.value /
                  trendAxisMax,
              ),
            );

          const y =
            105 -
            normalized * 95;

          return `${x},${y}`;
        },
      )
      .join(" ");
  }

  function getTrendPointPosition(
    value: number,
    index: number,
    length: number,
  ) {
    const x =
      length === 1
        ? 250
        : (
            index /
            (
              length -
              1
            )
          ) *
          500;

    const normalized =
      Math.max(
        0,
        Math.min(
          1,
          value /
            trendAxisMax,
        ),
      );

    const y =
      105 -
      normalized * 95;

    return {
      x,
      y,
    };
  }

  const trendPoints =
    createTrendPoints(
      trendData,
    );

  const previousTrendPoints =
    createTrendPoints(
      previousTrendData,
    );

  const currentTrendAverage =
    trendData.length > 0
      ? trendData.reduce(
          (total, point) =>
            total +
            point.value,
          0,
        ) /
        trendData.length
      : null;

  const previousTrendAverage =
    previousTrendData.length >
    0
      ? previousTrendData.reduce(
          (total, point) =>
            total +
            point.value,
          0,
        ) /
        previousTrendData.length
      : null;

  const trendInsight =
    currentTrendAverage === null
      ? `No recent ${trendMetric.toLowerCase()} data available.`
      : previousTrendAverage ===
          null
        ? `Current ${trendMetric.toLowerCase()} data is available, but there is not enough previous-period data for comparison.`
        : Math.abs(
              currentTrendAverage -
                previousTrendAverage,
            ) < 0.05
          ? `${trendMetric} is approximately unchanged compared with the previous 4 weeks.`
          : currentTrendAverage >
              previousTrendAverage
            ? `${trendMetric} is higher than during the previous 4 weeks.`
            : `${trendMetric} is lower than during the previous 4 weeks.`;
  const checkinInsight =
    useMemo(() => {
      const topDifficulty =
        unexpectedDifficulty[0];

      const topUnusual =
        unusualResponses[0];

      if (
        topDifficulty &&
        topUnusual
      ) {
        return `${topDifficulty.label} was harder than expected in ${topDifficulty.percentage}% of recent sessions. ${topUnusual.athlete} currently shows the strongest review signal: ${topUnusual.signal.toLowerCase()}.`;

      if (topDifficulty) {
        return `${topDifficulty.label} was harder than expected in ${topDifficulty.percentage}% of recent sessions across ${topDifficulty.athletes} athlete${topDifficulty.athletes === 1 ? "" : "s"}.`;

      if (topUnusual) {
        return `${topUnusual.athlete} currently shows a check-in signal worth reviewing: ${topUnusual.signal.toLowerCase()} (${topUnusual.detail.toLowerCase()}).`;
      }

      if (currentPeriodCheckins.length > 0) {
        return "Recent check-ins do not currently show a repeated difficulty or athlete-level review signal.";
      }

      return "No recent check-in data is available for this analysis period.";
    }, [
      unexpectedDifficulty,
      unusualResponses,
      currentPeriodCheckins,
    ]);

  const formatChange = (
    value: number | null,
    suffix = "",
  ) => {
    if (value === null) {
      return "No previous-period comparison";
    }

    if (value === 0) {
      return "No change vs previous 4 weeks";
    }

    const arrow = value > 0 ? "↑" : "↓";

    return `${arrow} ${Math.abs(value)}${suffix} vs previous 4 weeks`;
  };

  const factors =
    summary.performanceFactors
      .slice(0, 6)
      .map((factor) => ({
        ...factor,
        value: factor.percentage,
        icon:
          getPerformanceFactorIcon(
            factor.label,
          ),
      }));

  const difficultyEasierEnd =
    summary.difficulty.easier * 3.6;

  const difficultyExpectedEnd =
    difficultyEasierEnd +
    summary.difficulty.expected * 3.6;

  const difficultyHarderEnd =
    difficultyExpectedEnd +
    summary.difficulty.harder * 3.6;

  const difficultyMuchHarderEnd =
    Math.min(
      360,
      difficultyHarderEnd +
        summary.difficulty.muchHarder * 3.6,
    );

  const difficultyGradient =
    summary.totalSessions === 0
      ? "conic-gradient(#e2e8f0 0deg 360deg)"
      : `conic-gradient(
          #22c55e 0deg ${difficultyEasierEnd}deg,
          #3b82f6 ${difficultyEasierEnd}deg ${difficultyExpectedEnd}deg,
          #f97316 ${difficultyExpectedEnd}deg ${difficultyHarderEnd}deg,
          #ef4444 ${difficultyHarderEnd}deg ${difficultyMuchHarderEnd}deg,
          #e2e8f0 ${difficultyMuchHarderEnd}deg 360deg
        )`;

  const painChange =
    summary.painReports -
    comparison.previous.painReports;

  const stressChange =
    summary.stressReports -
    comparison.previous.stressReports;

  const formatCountChange = (
    value: number,
  ) => {
    if (value === 0) {
      return "No change vs previous period";
    }

    return `${
      value > 0 ? "↑" : "↓"
    } ${Math.abs(value)} vs previous period`;
  };

  const metrics = [
    {
      label: "Avg RPE",
      value:
        summary.averageRpe !== null
          ? summary.averageRpe.toFixed(1)
          : "—",
      suffix: "/10",
      detail: formatChange(
        comparison.changes.averageRpe,
      ),
      icon: CircleUserRound,
      style: "bg-orange-50 text-orange-600",
    },
    {
      label: "Goal Achieved",
      value:
        summary.goalAchievedPercentage !== null
          ? `${summary.goalAchievedPercentage}%`
          : "—",
      detail: formatChange(
        comparison.changes.goalAchieved,
        "%",
      ),
      icon: Target,
      style: "bg-green-50 text-green-600",
    },
    {
      label: "Stable / Improved",
      value:
        summary.stableOrImprovedPercentage !== null
          ? `${summary.stableOrImprovedPercentage}%`
          : "—",
      detail: formatChange(
        comparison.changes.stableOrImproved,
        "%",
      ),
      icon: Activity,
      style: "bg-blue-50 text-blue-600",
    },
    {
      label: "Sessions to Review",
      value: String(
        summary.sessionsToReview,
      ),
      detail:
        comparison.changes.sessionsToReview === 0
          ? "No change vs previous 4 weeks"
          : `${
              comparison.changes.sessionsToReview > 0
                ? "↑"
                : "↓"
            } ${Math.abs(
              comparison.changes.sessionsToReview,
            )} vs previous 4 weeks`,
      icon: TrendingDown,
      style: "bg-red-50 text-red-600",

  if (loading) {
    return (
      <div className="rounded-2xl border border-border bg-white p-6 text-center shadow-sm">
        <p className="text-sm font-semibold">
          Loading Check-In Analysis...

  if (error) {

  if (!data || data.athleteIds.length === 0) {

      .slice(0, 4);

  return (
    <div className="space-y-3">


      <section className="grid grid-cols-4 gap-2">
        {metrics.map(({ label, value, suffix, detail, icon: Icon, style }) => (
          <article
            key={label}
            className="rounded-2xl border border-border bg-white px-3 py-2 shadow-sm"
          >
            <div className="flex items-center gap-2.5">
              <div
                className={`grid h-8 w-8 shrink-0 place-items-center rounded-full ${style}`}
              >
                <Icon className="h-3.5 w-3.5" />
              </div>

              <div className="min-w-0">
                <p className="text-[8px] font-semibold text-muted-foreground">
                  {label}
                </p>

                <div className="mt-0.5 flex items-baseline gap-1">
                  <p className="text-xl font-bold">{value}</p>
                  {suffix && (
                    <span className="text-[9px] text-muted-foreground">
                      {suffix}
                    </span>
                  )}
                </div>

                <p className={`text-[7px] ${style.split(" ")[1]}`}>
                  {detail}
                </p>
              </div>
            </div>
          </article>
        ))}
      </section>

      <section className="rounded-2xl border border-border bg-white p-3 shadow-sm">
        <div className="flex items-start justify-between gap-3">
          <div>
            <p className="text-[8px] font-bold uppercase tracking-[0.14em] text-violet-600">
              Coach Priority
            </p>

            <h3 className="mt-1 text-sm font-bold">
              Which athletes should I look at?

            <p className="mt-1 text-[8px] text-muted-foreground">
              Recent athlete-level check-in signals, ordered for faster review.
            </p>
          </div>

          <span className="rounded-lg bg-violet-50 px-2.5 py-1 text-[8px] font-semibold text-violet-700">
            {unusualResponses.length} to review
          </span>
        </div>

        <div className="mt-3 grid grid-cols-2 gap-3">
          <div className="rounded-xl border border-orange-100 bg-orange-50/30 p-2.5">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-[9px] font-bold text-orange-700">
                  Review first
                </p>

                <p className="mt-0.5 text-[7px] text-muted-foreground">
                  Athletes with unusual recent responses
                </p>
              </div>

              <span className="rounded-full bg-orange-100 px-2 py-0.5 text-[7px] font-bold text-orange-700">
                {unusualResponses.length}
              </span>
            </div>

            <div className="mt-2 space-y-1.5">

                      <p className="mt-0.5 truncate text-[7px] text-muted-foreground">
                        {item.signal}

                      </span>

                      <ChevronRight className="h-3 w-3 text-muted-foreground" />

                  </button>

              {unusualResponses.length === 0 && (
                <div className="rounded-lg border border-dashed border-orange-100 bg-white px-3 py-4 text-center">
                  <p className="text-[8px] font-semibold">
                    No unusual athlete responses
                  </p>

                  <p className="mt-1 text-[7px] text-muted-foreground">
                    No athlete currently meets the review rules.
                  </p>
                </div>

            </div>
          </div>

            <div className="flex items-center justify-between">
              <div>
                <p className="text-[9px] font-bold text-green-700">
                  No current review signal
                </p>

                <p className="mt-0.5 text-[7px] text-muted-foreground">
                  No unusual response detected in this analysis
                </p>
              </div>

              <span className="rounded-full bg-green-100 px-2 py-0.5 text-[7px] font-bold text-green-700">
                {athletesWithoutReviewSignal.length}
              </span>
            </div>

            <div className="mt-2 space-y-1.5">
              {athletesWithoutReviewSignal.map(
                (athlete) => (
                  <button
                    key={athlete.id}
                    type="button"
                    onClick={() =>
                      navigate({
                        to: "/coach-manager/athletes/$athleteId",
                        params: {
                          athleteId:
                            athlete.id,
                        },
                        search: {
                          from: "checkin",
                        },
                      })
                    }
                    className="flex w-full items-center justify-between rounded-lg border border-green-100 bg-white px-2.5 py-2 text-left transition hover:border-green-200 hover:bg-green-50/50"
                  >
                    <div>
                      <p className="text-[9px] font-semibold">
                        {athlete.display_name ??
                          "Unnamed athlete"}
                      </p>

                      <p className="mt-0.5 text-[7px] text-muted-foreground">
                        No current unusual-response signal
                      </p>
                    </div>

                    <ChevronRight className="h-3 w-3 text-muted-foreground" />
                  </button>
                ),
              )}

              {athletesWithoutReviewSignal.length ===
                0 && (
                <div className="rounded-lg border border-dashed border-green-100 bg-white px-3 py-4 text-center">
                  <p className="text-[8px] font-semibold">
                    All athletes currently have a review signal
                  </p>
                </div>
              )}
            </div>
          </div>
        </div>

        <p className="mt-2 text-[7px] text-muted-foreground">
          Priority reflects athlete-reported check-in patterns. It is a review aid, not a diagnosis or automatic coaching decision.
        </p>

      <div className="grid grid-cols-[1fr_0.9fr_1.12fr] gap-3">
        <section className="rounded-2xl border border-border bg-white p-2.5 shadow-sm">
          <div className="flex items-start justify-between gap-2">
            <div>
              <h3 className="text-xs font-bold">
                1. How are athletes responding?
              </h3>

              <p className="mt-1 text-[8px] text-muted-foreground">
                Trends across the selected period
              </p>
            </div>

            <div className="flex rounded-lg bg-muted/40 p-0.5">
              <button
                type="button"
                onClick={() => setTrendMode("average")}
                className={[
                  "rounded-md px-2 py-1 text-[7px] font-semibold",
                  trendMode === "average"
                    ? "bg-white text-violet-700 shadow-sm"
                    : "text-muted-foreground",
                ].join(" ")}
              >
                Team Average
              </button>

              <button
                type="button"
                onClick={() => setTrendMode("unusual")}
                className={[
                  "rounded-md px-2 py-1 text-[7px] font-semibold",
                  trendMode === "unusual"
                    ? "bg-white text-violet-700 shadow-sm"
                    : "text-muted-foreground",
                ].join(" ")}
              >
                Unusual Responses
              </button>
            </div>
          </div>

          <div className="mt-3 flex gap-1">
            {(["RPE", "Physical Feeling", "Performance Quality", "Goal Achievement"] as const).map(
              (item) => (
                <button
                  key={item}
                  type="button"
                  onClick={() => setTrendMetric(item)}
                  className={[
                    "rounded-lg border px-2 py-1 text-[7px] font-semibold",
                    trendMetric === item
                      ? "border-violet-200 bg-violet-50 text-violet-700"
                      : "border-border text-muted-foreground",
                  ].join(" ")}
                >
                  {item}
                </button>
              ),
            )}
          </div>

          {trendMode === "average" ? (
            <>
              <div className="relative mt-3 h-36">
                <div className="absolute inset-x-0 top-0 bottom-5 flex flex-col justify-between">
                  {trendAxisLabels.map((value) => (
                    <div key={value} className="flex items-center gap-2">
                      <span className="w-5 text-[7px] text-muted-foreground">
                        {value}
                      </span>

                      <div className="h-px flex-1 bg-border/50" />
                    </div>
                  ))}
                </div>

                <svg
                  viewBox="0 0 500 120"
                  className="absolute left-6 top-1 h-[105px] w-[calc(100%-24px)]"
                  preserveAspectRatio="none"
                >
                  {previousTrendPoints && (
                    <polyline
                      points={
                        previousTrendPoints
                      }
                      fill="none"
                      stroke="currentColor"
                      strokeWidth="1.5"

                      strokeLinecap="round"
                      strokeLinejoin="round"
                      className="text-slate-300"
                    />
                  )}

                  {trendPoints && (
                    <polyline
                      points={
                        trendPoints
                      }
                      fill="none"
                      stroke="currentColor"
                      strokeWidth="2.4"
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      className="text-violet-600"
                    />
                  )}

                  {trendData.map(

                    <span>No recent {trendMetric} data</span>

              <div className="mt-2 flex items-center gap-4 text-[7px] text-muted-foreground">
                <span className="flex items-center gap-1">

function getPerformanceFactorIcon(
  label: string,
) {
  const normalized =
    label.toLowerCase();

  if (
    normalized.includes("schlaf") ||
    normalized.includes("sleep")
  ) {
    return MoonStar;
  }

  if (
    normalized.includes("stress") ||
    normalized.includes("druck") ||
    normalized.includes("pressure")
  ) {
    return Brain;
  }

  if (
    normalized.includes("schule") ||
    normalized.includes("school") ||
    normalized.includes("arbeit") ||
    normalized.includes("work")
  ) {
    return Activity;
  }

  if (
    normalized.includes("wetter") ||
    normalized.includes("weather")
  ) {
    return Zap;
  }

  if (
    normalized.includes("training")
  ) {
    return Target;
  }

  if (
    normalized.includes("wettkampf") ||
    normalized.includes("competition")
  ) {
    return Activity;
  }

  return Activity;
}

  }

  const athleteReviewIds =
    new Set(

  const athletesWithoutReviewSignal =

            </p>

            <h3 className="mt-1 text-sm font-bold">
              Which athletes should I look at?
            </h3>

            <p className="mt-1 text-[8px] text-muted-foreground">
              Recent athlete-level check-in signals, ordered for faster review.
            </p>
          </div>

          <span className="rounded-lg bg-violet-50 px-2.5 py-1 text-[8px] font-semibold text-violet-700">
            {unusualResponses.length} to review
          </span>
        </div>

        <div className="mt-3 grid grid-cols-2 gap-3">
          <div className="rounded-xl border border-orange-100 bg-orange-50/30 p-2.5">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-[9px] font-bold text-orange-700">
                  Review first
                </p>

                <p className="mt-0.5 text-[7px] text-muted-foreground">
                  Athletes with unusual recent responses

              <span className="rounded-full bg-orange-100 px-2 py-0.5 text-[7px] font-bold text-orange-700">

                  onClick={() => setTrendMetric(item)}

                  {item}

          </div>

          {trendMode === "average" ? (
            <>
              <div className="relative mt-3 h-36">
                <div className="absolute inset-x-0 top-0 bottom-5 flex flex-col justify-between">
                  {trendAxisLabels.map((value) => (
                    <div key={value} className="flex items-center gap-2">

                      </span>

                      <div className="h-px flex-1 bg-border/50" />

                </div>

                      strokeLinecap="round"

                        trendPoints
                      }
                      fill="none"
                      stroke="currentColor"
                      strokeWidth="2.4"
                      strokeLinecap="round"
                      strokeLinejoin="round"

                    />

                        y,
                      } =
                        getTrendPointPosition(
                          point.value,
                          index,

                        );

                        <circle
                          key={
                            point.date
                          }
                          cx={x}
                          cy={y}
                          r="2.8"
                          fill="white"
                          stroke="currentColor"
                          strokeWidth="2"
                          className="text-violet-600"
                        />
                      );
                    },
                  )}
                </svg>

                <div className="absolute bottom-0 left-6 right-0 flex justify-between text-[7px] text-muted-foreground">
                  {trendData.length > 0 ? (
                    trendData.map((point) => (
                      <span key={point.date}>
                        {point.label}
                      </span>
                    ))
                  ) : (
                    <span>No recent {trendMetric} data</span>
                  )}
                </div>
              </div>

                <span className="flex items-center gap-1">
                  <span className="h-0.5 w-4 bg-violet-600" />
                  Current 4 weeks
                </span>

                <span className="flex items-center gap-1">
                  <span className="w-4 border-t border-dashed border-slate-400" />
                  Previous 4 weeks
                </span>
              </div>

              <div className="mt-3 rounded-lg bg-violet-50 px-2.5 py-2 text-[8px] text-violet-700">
                {trendInsight}
              </div>
            </>
          ) : (

              <div className="rounded-lg border border-orange-100 bg-orange-50/40 px-2.5 py-2">
                <p className="text-[8px] font-semibold text-orange-700">
                  {summary.mostReportedPainArea

                <p className="mt-0.5 text-[7px] text-muted-foreground">
                  These are changes worth inspecting, not automatic problems.

                  onClick={() =>
                    navigate({
                      to: "/coach-manager/athletes/$athleteId",

                        from: "checkin",

                      {item.athlete}

                      {item.signal}
                    </p>

                  <div className="flex items-center gap-2">
                    <span
                      className={`rounded-full px-2 py-1 text-[7px] font-semibold ${item.style}`}
                    >
                      {item.detail}
                    </span>

                    <ChevronRight className="h-3 w-3 text-muted-foreground" />

            2. Was training harder than expected?
          </h3>

          <p className="mt-1 text-[8px] text-muted-foreground">
            Intended difficulty compared with athlete perception
          </p>

          <div className="mt-4 grid grid-cols-[110px_1fr] items-center gap-4">
            <div
              className="relative grid h-24 w-24 place-items-center rounded-full"
              style={{
                background:
                  difficultyGradient,
              }}
            >
              <div className="grid h-14 w-14 place-items-center rounded-full bg-white text-center">
                <div>
                  <p className="text-base font-bold">
                  {summary.totalSessions}

                  <p className="text-[7px] text-muted-foreground">
                    Sessions
                  </p>

            </div>

          </div>

            Where is unexpected difficulty happening?
          </p>

            {unexpectedDifficulty.map((item) => (

                key={item.label}

                  {item.percentage}%
                </span>

                <span className="text-[7px] text-muted-foreground">

            Session evidence shown above
          </p>
        </section>

            3. Did training work as intended?
          </h3>

          <p className="mt-1 text-[8px] text-muted-foreground">
            Simple session outcome based on goal achievement and performance quality

          {(() => {

              Math.max(
                0,
                summary.totalSessions -
                  workedWell -
                  review,

              workedWell + mixed + review;

              total > 0
                ? Math.round((count / total) * 100)
                : 0;

            return (
              <>
                <div className="mt-3 grid grid-cols-3 gap-2">
                  <div className="rounded-xl border border-green-100 bg-green-50/60 p-2.5">
                    <p className="text-[8px] font-bold text-green-700">
                      Worked well
                    </p>

                    <p className="mt-1 text-xl font-bold text-green-700">
                      {percentage(workedWell)}%
                    </p>

                    <p className="text-[7px] text-muted-foreground">
                      {workedWell} sessions
                    </p>
                  </div>

                  <div className="rounded-xl border border-blue-100 bg-blue-50/60 p-2.5">
                    <p className="text-[8px] font-bold text-blue-700">
                      Mixed
                    </p>

                    <p className="mt-1 text-xl font-bold text-blue-700">
                      {percentage(mixed)}%
                    </p>

                    <p className="text-[7px] text-muted-foreground">
                      {mixed} sessions
                    </p>
                  </div>

                  <div className="rounded-xl border border-red-100 bg-red-50/60 p-2.5">
                    <p className="text-[8px] font-bold text-red-700">
                      Review
                    </p>

                    <p className="mt-1 text-xl font-bold text-red-700">
                      {percentage(review)}%
                    </p>

                    <p className="text-[7px] text-muted-foreground">
                      {review} sessions
                    </p>
                  </div>
                </div>

                <div className="mt-2 rounded-lg bg-muted/30 px-2.5 py-2">
                  <p className="text-[7px] leading-relaxed text-muted-foreground">
                    <span className="font-semibold text-foreground">
                      Worked well
                    </span>{" "}
                    = goal achieved with stable or improved performance.{" "}
                    <span className="font-semibold text-foreground">
                      Review
                    </span>{" "}
                    = goal missed or performance declined. Other sessions are shown as mixed.
                  </p>
                </div>
              </>
            );
          })()}
        </section>
      </div>

      <div className="grid grid-cols-2 gap-3">
        <section className="rounded-2xl border border-border bg-white p-2.5 shadow-sm">
          <h3 className="text-xs font-bold">
            4. Frequently Reported Performance Factors
          </h3>

          <p className="mt-1 text-[8px] text-muted-foreground">
            Factors athletes say influenced their performance
          </p>

          <div className="mt-2.5 space-y-1.5">

                </p>
              </div>
            )}

            {factors.map(({ label, value, count, icon: Icon }) => (
              <div
                key={label}
                className="grid grid-cols-[20px_105px_1fr_32px] items-center gap-2"
              >
                <div className="grid h-5 w-5 place-items-center rounded-md bg-violet-50">
                  <Icon className="h-3 w-3 text-violet-600" />
                </div>

                <span className="text-[8px] font-medium">{label}</span>

                      width: `${Math.min(value, 100)}%`,

                    className="h-full rounded-full bg-violet-500"

                <span
                  className="text-right text-[8px] font-semibold"
                  title={`${count} sessions`}
                >
                  {value}%
                </span>
              </div>
            ))}
          </div>

          <p className="mt-3 text-[8px] font-semibold text-violet-600">
            Factor summary shown above
          </p>
        </section>

        <section className="rounded-2xl border border-border bg-white p-2.5 shadow-sm">
          <h3 className="text-xs font-bold">
            5. Body & Mental Signals
          </h3>

          <p className="mt-1 text-[8px] text-muted-foreground">
            Important signals reported through check-ins
          </p>

          <div className="mt-2.5 grid grid-cols-3 gap-2">
            <div className="rounded-xl border border-red-100 bg-red-50/40 p-2.5">
              <HeartPulse className="h-4 w-4 text-red-500" />
              <p className="mt-2 text-lg font-bold">
                {summary.painReports}
              </p>
              <p className="text-[8px] font-semibold">Pain reports</p>
              <p className="mt-1 text-[7px] text-muted-foreground">
                {formatCountChange(

              <p className="mt-2 text-lg font-bold">
                {summary.stressReports}

              <p className="mt-1 text-[7px] text-muted-foreground">
                {formatCountChange(
                  stressChange,
                )}
              </p>
            </div>

            <div className="rounded-xl border border-orange-100 bg-orange-50/40 p-2.5">
              <Activity className="h-4 w-4 text-orange-600" />
              <p className="mt-2 text-sm font-bold">
                {summary.mostReportedPainArea?.area ?? "—"}
              </p>
              <p className="text-[8px] font-semibold">Most reported area</p>

                {summary.mostReportedPainArea

            </div>
          </div>

          {(summary.painReports > 0 ||
            summary.stressReports > 0) && (
            <div className="mt-1.5 rounded-lg border border-orange-100 bg-orange-50/50 p-1.5 text-[7px] text-orange-800">
              {summary.painReports > 0 &&
              summary.stressReports > 0
                ? `${summary.painReports} pain reports and ${summary.stressReports} stress-related reports were recorded during the current period.`
                : summary.painReports > 0
                  ? `${summary.painReports} pain reports were recorded during the current period.`
                  : `${summary.stressReports} stress-related reports were recorded during the current period.`}
            </div>
          )}
        </section>


      </div>


      <section className="flex items-center gap-3 rounded-2xl border border-violet-100 bg-gradient-to-r from-violet-50/70 via-white to-orange-50/30 p-3 shadow-sm">
        <div className="grid h-9 w-9 shrink-0 place-items-center rounded-xl bg-violet-100">
          <Sparkles className="h-4 w-4 text-violet-600" />
        </div>

        <div className="min-w-0 flex-1">
          <p className="text-[10px] font-bold text-violet-700">
            AI Coach Insight
          </p>

          <p className="mt-1 text-[8px] leading-relaxed text-muted-foreground">
            {checkinInsight}
          </p>
        </div>

        <span className="rounded-xl border border-violet-200 bg-white px-3 py-2 text-[8px] font-semibold text-violet-700">
          AI insight shown here
        </span>
      </section>
    </div>
  );
}

import { useEffect, useMemo, useState } from "react";

import {
  getCoachCheckinData,
  type CoachCheckinData,
} from "./CheckinAnalysisData";

import {
  compareCheckinPeriods,
} from "./CheckinAnalysisComparisons";

import {
  calculateCheckinTrend,
  calculateGoalPerformanceMatrix,
  calculateUnexpectedDifficulty,
  calculateUnusualResponses,
} from "./CheckinAnalysisCalculations";
import {
  Activity,
  Brain,
  ChevronRight,
  CircleUserRound,
  HeartPulse,
  MoonStar,
  Sparkles,
  Target,
  TrendingDown,
  Users,
  Zap,
} from "lucide-react";


function getPerformanceFactorIcon(
  label: string,
) {
  const normalized =
    label.toLowerCase();

  if (
    normalized.includes("schlaf") ||
    normalized.includes("sleep")
  ) {
    return MoonStar;
  }

  if (
    normalized.includes("stress") ||
    normalized.includes("druck") ||
    normalized.includes("pressure")
  ) {
    return Brain;
  }

  if (
    normalized.includes("schule") ||
    normalized.includes("school") ||
    normalized.includes("arbeit") ||
    normalized.includes("work")
  ) {
    return Activity;
  }

  if (
    normalized.includes("wetter") ||
    normalized.includes("weather")
  ) {
    return Zap;
  }

  if (
    normalized.includes("training")
  ) {
    return Target;
  }

  if (
    normalized.includes("wettkampf") ||
    normalized.includes("competition")
  ) {
    return Activity;
  }

  return Activity;
}

export function CheckinAnalysis() {
  const navigate = useNavigate();
  const [trendMetric, setTrendMetric] = useState<"RPE" | "Physical Feeling" | "Performance Quality" | "Goal Achievement">("RPE");
  const [trendMode, setTrendMode] =
    useState<"average" | "unusual">("average");

  const [showOutcomeReview, setShowOutcomeReview] =
    useState(false);

  const [data, setData] =
    useState<CoachCheckinData | null>(null);

  const [loading, setLoading] =
    useState(true);

  const [error, setError] =
    useState<string | null>(null);

  useEffect(() => {
    let cancelled = false;

    async function loadCheckins() {
      try {
        setLoading(true);
        setError(null);

        const result =
          await getCoachCheckinData();

        if (!cancelled) {
          setData(result);
        }
      } catch (loadError) {
        console.error(
          "Could not load Check-In Analysis:",
          loadError,
        );

        if (!cancelled) {
          setError(
            loadError instanceof Error
              ? loadError.message
              : "Could not load check-in analysis.",
          );
        }
      } finally {
        if (!cancelled) {
          setLoading(false);
        }
      }
    }

    void loadCheckins();

    return () => {
      cancelled = true;
    };
  }, []);

  const comparison = useMemo(
    () =>
      compareCheckinPeriods(
        data?.checkins ?? [],
      ),
    [data],
  );

  const summary = comparison.current;

  /*
  ============================================================
  CURRENT 4-WEEK RAW ROWS

  These rows power sections that need session-level data rather
  than only the summary totals.
  ============================================================
  */
  const currentPeriodCheckins =
    useMemo(
      () => {
        const today =
          new Date();

        today.setHours(
          0,
          0,
          0,
          0,
        );

        const start =
          new Date(today);

        start.setDate(
          start.getDate() -
            27,
        );

        return (
          data?.checkins ?? []
        ).filter(
          (checkin) => {
            const date =
              new Date(
                `${checkin.checkin_date}T00:00:00`,
              );

            return (
              date >= start &&
              date <= today &&
              checkin.is_rest_day !== true
            );
          },
        );
      },
      [data],
    );

  const previousPeriodCheckins =
    useMemo(
      () => {
        const today =
          new Date();

        today.setHours(
          0,
          0,
          0,
          0,
        );

        const previousEnd =
          new Date(today);

        previousEnd.setDate(
          previousEnd.getDate() -
            28,
        );

        const previousStart =
          new Date(today);

        previousStart.setDate(
          previousStart.getDate() -
            55,
        );

        return (
          data?.checkins ?? []
        ).filter(
          (checkin) => {
            const date =
              new Date(
                `${checkin.checkin_date.slice(
                  0,
                  10,
                )}T00:00:00`,
              );

            return (
              date >= previousStart &&
              date <= previousEnd
            );
          },
        );
      },
      [data],
    );
  const unexpectedDifficulty =
    useMemo(
      () =>
        calculateUnexpectedDifficulty(
          currentPeriodCheckins,
        ),
      [currentPeriodCheckins],
    );

  const matrix =
    useMemo(
      () =>
        calculateGoalPerformanceMatrix(
          currentPeriodCheckins,
        ),
      [currentPeriodCheckins],
    );

  const athleteNameById =
    useMemo(
      () =>
        new Map(
          (data?.athletes ?? []).map(
            (athlete) => [
              athlete.id,
              athlete.display_name ??
                "Unnamed athlete",
            ],
          ),
        ),
      [data],
    );

  const outcomeReviewSessions =
    useMemo(
      () =>
        currentPeriodCheckins
          .filter(
            (checkin) =>
              checkin.session_goal_achieved === "Nein" ||
              checkin.performance_quality ===
                "Stark abgefallen",
          )
          .map((checkin) => {
            const reasons: string[] = [];

            if (
              checkin.session_goal_achieved ===
              "Nein"
            ) {
              reasons.push(
                "Goal not achieved",
              );
            }

            if (
              checkin.performance_quality ===
              "Stark abgefallen"
            ) {
              reasons.push(
                "Performance strongly declined",
              );
            }

            return {
              athleteId:
                checkin.user_id,

              athlete:
                athleteNameById.get(
                  checkin.user_id,
                ) ??
                "Unknown athlete",

              date:
                checkin.checkin_date,

              workoutType:
                typeof checkin.workout_type ===
                  "string" &&
                checkin.workout_type.trim()
                  .length > 0
                  ? checkin.workout_type.trim()
                  : "Training",

              reason:
                reasons.join(" · "),
            };
          })
          .sort(
            (a, b) =>
              new Date(b.date).getTime() -
              new Date(a.date).getTime(),
          ),
      [
        currentPeriodCheckins,
        athleteNameById,
      ],
    );
  const unusualResponses =
    useMemo(
      () =>
        calculateUnusualResponses(
          currentPeriodCheckins,
        ).map((item) => ({
          ...item,
          athlete:
            athleteNameById.get(
              item.athleteId,
            ) ?? "Unknown athlete",
        })),
      [
        currentPeriodCheckins,
        athleteNameById,
      ],
    );

  const trendData =
    useMemo(
      () =>
        calculateCheckinTrend(
          currentPeriodCheckins,
          trendMetric,
        ),
      [
        currentPeriodCheckins,
        trendMetric,
      ],
    );

  const previousTrendData =
    useMemo(
      () =>
        calculateCheckinTrend(
          previousPeriodCheckins,
          trendMetric,
        ),
      [
        previousPeriodCheckins,
        trendMetric,
      ],
    );

  const trendAxisMax =
    trendMetric === "RPE"
      ? 10
      : trendMetric ===
          "Physical Feeling"
        ? 5
        : trendMetric ===
            "Performance Quality"
          ? 4
          : 3;

  const trendAxisLabels =
    [
      trendAxisMax,
      trendAxisMax * 0.75,
      trendAxisMax * 0.5,
      trendAxisMax * 0.25,
      0,
    ].map(
      (value) =>
        Number.isInteger(value)
          ? value
          : Number(
              value.toFixed(1),
            ),
    );

  function createTrendPoints(
    points:
      typeof trendData,
  ) {
    if (
      points.length === 0
    ) {
      return "";
    }

    return points
      .map(
        (point, index) => {
          const x =
            points.length === 1
              ? 250
              : (
                  index /
                  (
                    points.length -
                    1
                  )
                ) *
                500;

          const normalized =
            Math.max(
              0,
              Math.min(
                1,
                point.value /
                  trendAxisMax,
              ),
            );

          const y =
            105 -
            normalized * 95;

          return `${x},${y}`;
        },
      )
      .join(" ");
  }

  function getTrendPointPosition(
    value: number,
    index: number,
    length: number,
  ) {
    const x =
      length === 1
        ? 250
        : (
            index /
            (
              length -
              1
            )
          ) *
          500;

    const normalized =
      Math.max(
        0,
        Math.min(
          1,
          value /
            trendAxisMax,
        ),
      );

    const y =
      105 -
      normalized * 95;

    return {
      x,
      y,
    };
  }

  const trendPoints =
    createTrendPoints(
      trendData,
    );

  const previousTrendPoints =
    createTrendPoints(
      previousTrendData,
    );

  const currentTrendAverage =
    trendData.length > 0
      ? trendData.reduce(
          (total, point) =>
            total +
            point.value,
          0,
        ) /
        trendData.length
      : null;

  const previousTrendAverage =
    previousTrendData.length >
    0
      ? previousTrendData.reduce(
          (total, point) =>
            total +
            point.value,
          0,
        ) /
        previousTrendData.length
      : null;

  const trendInsight =
    currentTrendAverage === null
      ? `No recent ${trendMetric.toLowerCase()} data available.`
      : previousTrendAverage ===
          null
        ? `Current ${trendMetric.toLowerCase()} data is available, but there is not enough previous-period data for comparison.`
        : Math.abs(
              currentTrendAverage -
                previousTrendAverage,
            ) < 0.05
          ? `${trendMetric} is approximately unchanged compared with the previous 4 weeks.`
          : currentTrendAverage >
              previousTrendAverage
            ? `${trendMetric} is higher than during the previous 4 weeks.`
            : `${trendMetric} is lower than during the previous 4 weeks.`;
  const checkinInsight =
    useMemo(() => {
      const topDifficulty =
        unexpectedDifficulty[0];

      const topUnusual =
        unusualResponses[0];

      if (
        topDifficulty &&
        topUnusual
      ) {
        return `${topDifficulty.label} was harder than expected in ${topDifficulty.percentage}% of recent sessions. ${topUnusual.athlete} currently shows the strongest review signal: ${topUnusual.signal.toLowerCase()}.`;
      }

      if (topDifficulty) {
        return `${topDifficulty.label} was harder than expected in ${topDifficulty.percentage}% of recent sessions across ${topDifficulty.athletes} athlete${topDifficulty.athletes === 1 ? "" : "s"}.`;
      }

      if (topUnusual) {
        return `${topUnusual.athlete} currently shows a check-in signal worth reviewing: ${topUnusual.signal.toLowerCase()} (${topUnusual.detail.toLowerCase()}).`;
      }

      if (currentPeriodCheckins.length > 0) {
        return "Recent check-ins do not currently show a repeated difficulty or athlete-level review signal.";
      }

      return "No recent check-in data is available for this analysis period.";
    }, [
      unexpectedDifficulty,
      unusualResponses,
      currentPeriodCheckins,
    ]);

  const formatChange = (
    value: number | null,
    suffix = "",
  ) => {
    if (value === null) {
      return "No previous-period comparison";
    }

    if (value === 0) {
      return "No change vs previous 4 weeks";
    }

    const arrow = value > 0 ? "↑" : "↓";

    return `${arrow} ${Math.abs(value)}${suffix} vs previous 4 weeks`;
  };

  const factors =
    summary.performanceFactors
      .slice(0, 6)
      .map((factor) => ({
        ...factor,
        value: factor.percentage,
        icon:
          getPerformanceFactorIcon(
            factor.label,
          ),
      }));

  const difficultyEasierEnd =
    summary.difficulty.easier * 3.6;

  const difficultyExpectedEnd =
    difficultyEasierEnd +
    summary.difficulty.expected * 3.6;

  const difficultyHarderEnd =
    difficultyExpectedEnd +
    summary.difficulty.harder * 3.6;

  const difficultyMuchHarderEnd =
    Math.min(
      360,
      difficultyHarderEnd +
        summary.difficulty.muchHarder * 3.6,
    );

  const difficultyGradient =
    summary.totalSessions === 0
      ? "conic-gradient(#e2e8f0 0deg 360deg)"
      : `conic-gradient(
          #22c55e 0deg ${difficultyEasierEnd}deg,
          #3b82f6 ${difficultyEasierEnd}deg ${difficultyExpectedEnd}deg,
          #f97316 ${difficultyExpectedEnd}deg ${difficultyHarderEnd}deg,
          #ef4444 ${difficultyHarderEnd}deg ${difficultyMuchHarderEnd}deg,
          #e2e8f0 ${difficultyMuchHarderEnd}deg 360deg
        )`;

  const painChange =
    summary.painReports -
    comparison.previous.painReports;

  const stressChange =
    summary.stressReports -
    comparison.previous.stressReports;

  const formatCountChange = (
    value: number,
  ) => {
    if (value === 0) {
      return "No change vs previous period";
    }

    return `${
      value > 0 ? "↑" : "↓"
    } ${Math.abs(value)} vs previous period`;
  };

  const metrics = [
    {
      label: "Avg RPE",
      value:
        summary.averageRpe !== null
          ? summary.averageRpe.toFixed(1)
          : "—",
      suffix: "/10",
      detail: formatChange(
        comparison.changes.averageRpe,
      ),
      icon: CircleUserRound,
      style: "bg-orange-50 text-orange-600",
    },
    {
      label: "Goal Achieved",
      value:
        summary.goalAchievedPercentage !== null
          ? `${summary.goalAchievedPercentage}%`
          : "—",
      detail: formatChange(
        comparison.changes.goalAchieved,
        "%",
      ),
      icon: Target,
      style: "bg-green-50 text-green-600",
    },
    {
      label: "Stable / Improved",
      value:
        summary.stableOrImprovedPercentage !== null
          ? `${summary.stableOrImprovedPercentage}%`
          : "—",
      detail: formatChange(
        comparison.changes.stableOrImproved,
        "%",
      ),
      icon: Activity,
      style: "bg-blue-50 text-blue-600",
    },
    {
      label: "Sessions to Review",
      value: String(
        summary.sessionsToReview,
      ),
      detail:
        comparison.changes.sessionsToReview === 0
          ? "No change vs previous 4 weeks"
          : `${
              comparison.changes.sessionsToReview > 0
                ? "↑"
                : "↓"
            } ${Math.abs(
              comparison.changes.sessionsToReview,
            )} vs previous 4 weeks`,
      icon: TrendingDown,
      style: "bg-red-50 text-red-600",
    },
  ];

  if (loading) {
    return (
      <div className="rounded-2xl border border-border bg-white p-6 text-center shadow-sm">
        <p className="text-sm font-semibold">
          Loading Check-In Analysis...
        </p>

        <p className="mt-1 text-xs text-muted-foreground">
          Reading athlete check-ins from Supabase.
        </p>
      </div>
    );
  }

  if (error) {
    return (
      <div className="rounded-2xl border border-red-200 bg-red-50 p-4">
        <p className="text-sm font-bold text-red-700">
          Check-In Analysis could not be loaded
        </p>

        <p className="mt-1 text-xs text-red-600">
          {error}
        </p>
      </div>
    );
  }

  if (!data || data.athleteIds.length === 0) {
    return (
      <div className="rounded-2xl border border-border bg-white p-6 text-center shadow-sm">
        <div className="mb-4 rounded-lg bg-yellow-50 p-3 text-left text-xs text-yellow-900">
          <p>DEBUG athleteIds: {JSON.stringify(data?.athleteIds ?? [])}</p>
          <p>DEBUG athletes: {JSON.stringify(data?.athletes ?? [])}</p>
          <p>DEBUG checkins: {data?.checkins.length ?? 0}</p>
        </div>
        <p className="text-sm font-bold">
          No athletes connected
        </p>

        <p className="mt-1 text-xs text-muted-foreground">
          Connect athletes to this coach before analysing check-ins.
        </p>
      </div>
    );
  }

  const athleteReviewIds =
    new Set(
      unusualResponses.map(
        (item) => item.athleteId,
      ),
    );

  const athletesWithoutReviewSignal =
    (data?.athletes ?? [])
      .filter(
        (athlete) =>
          !athleteReviewIds.has(
            athlete.id,
          ),
      )
      .slice(0, 4);

  return (
    <div className="space-y-3">


      <section className="grid grid-cols-4 gap-2">
        {metrics.map(({ label, value, suffix, detail, icon: Icon, style }) => (
          <article
            key={label}
            className="rounded-2xl border border-border bg-white px-3 py-2 shadow-sm"
          >
            <div className="flex items-center gap-2.5">
              <div
                className={`grid h-8 w-8 shrink-0 place-items-center rounded-full ${style}`}
              >
                <Icon className="h-3.5 w-3.5" />
              </div>

              <div className="min-w-0">
                <p className="text-[8px] font-semibold text-muted-foreground">
                  {label}
                </p>

                <div className="mt-0.5 flex items-baseline gap-1">
                  <p className="text-xl font-bold">{value}</p>
                  {suffix && (
                    <span className="text-[9px] text-muted-foreground">
                      {suffix}
                    </span>
                  )}
                </div>

                <p className={`text-[7px] ${style.split(" ")[1]}`}>
                  {detail}
                </p>
              </div>
            </div>
          </article>
        ))}
      </section>

      <section className="rounded-2xl border border-border bg-white p-3 shadow-sm">
        <div className="flex items-start justify-between gap-3">
          <div>
            <p className="text-[8px] font-bold uppercase tracking-[0.14em] text-violet-600">
              Coach Priority
            </p>

            <h3 className="mt-1 text-sm font-bold">
              Which athletes should I look at?
            </h3>

            <p className="mt-1 text-[8px] text-muted-foreground">
              Recent athlete-level check-in signals, ordered for faster review.
            </p>
          </div>

          <span className="rounded-lg bg-violet-50 px-2.5 py-1 text-[8px] font-semibold text-violet-700">
            {unusualResponses.length} to review
          </span>
        </div>

        <div className="mt-3 grid grid-cols-2 gap-3">
          <div className="rounded-xl border border-orange-100 bg-orange-50/30 p-2.5">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-[9px] font-bold text-orange-700">
                  Review first
                </p>

                <p className="mt-0.5 text-[7px] text-muted-foreground">
                  Athletes with unusual recent responses
                </p>
              </div>

              <span className="rounded-full bg-orange-100 px-2 py-0.5 text-[7px] font-bold text-orange-700">
                {unusualResponses.length}
              </span>
            </div>

            <div className="mt-2 space-y-1.5">
              {unusualResponses
                .slice(0, 4)
                .map((item) => (
                  <button
                    key={item.athleteId}
                    type="button"
                    onClick={() =>
                      navigate({
                        to: "/coach-manager/athletes/$athleteId",
                        params: {
                          athleteId:
                            item.athleteId,
                        },
                        search: {
                          from: "checkin",
                        },
                      })
                    }
                    className="flex w-full items-center justify-between rounded-lg border border-orange-100 bg-white px-2.5 py-2 text-left transition hover:border-orange-200 hover:bg-orange-50/50"
                  >
                    <div className="min-w-0">
                      <p className="truncate text-[9px] font-semibold">
                        {item.athlete}
                      </p>

                      <p className="mt-0.5 truncate text-[7px] text-muted-foreground">
                        {item.signal}
                      </p>
                    </div>

                    <div className="ml-2 flex shrink-0 items-center gap-2">
                      <span
                        className={`rounded-full px-2 py-0.5 text-[7px] font-semibold ${item.style}`}
                      >
                        {item.detail}
                      </span>

                      <ChevronRight className="h-3 w-3 text-muted-foreground" />
                    </div>
                  </button>
                ))}

              {unusualResponses.length === 0 && (
                <div className="rounded-lg border border-dashed border-orange-100 bg-white px-3 py-4 text-center">
                  <p className="text-[8px] font-semibold">
                    No unusual athlete responses
                  </p>

                  <p className="mt-1 text-[7px] text-muted-foreground">
                    No athlete currently meets the review rules.
                  </p>
                </div>
              )}
            </div>
          </div>

          <div className="rounded-xl border border-green-100 bg-green-50/30 p-2.5">
            <div className="flex items-center justify-between">
              <div>
                <p className="text-[9px] font-bold text-green-700">
                  No current review signal
                </p>

                <p className="mt-0.5 text-[7px] text-muted-foreground">
                  No unusual response detected in this analysis
                </p>
              </div>

              <span className="rounded-full bg-green-100 px-2 py-0.5 text-[7px] font-bold text-green-700">
                {athletesWithoutReviewSignal.length}
              </span>
            </div>

            <div className="mt-2 space-y-1.5">
              {athletesWithoutReviewSignal.map(
                (athlete) => (
                  <button
                    key={athlete.id}
                    type="button"
                    onClick={() =>
                      navigate({
                        to: "/coach-manager/athletes/$athleteId",
                        params: {
                          athleteId:
                            athlete.id,
                        },
                        search: {
                          from: "checkin",
                        },
                      })
                    }
                    className="flex w-full items-center justify-between rounded-lg border border-green-100 bg-white px-2.5 py-2 text-left transition hover:border-green-200 hover:bg-green-50/50"
                  >
                    <div>
                      <p className="text-[9px] font-semibold">
                        {athlete.display_name ??
                          "Unnamed athlete"}
                      </p>

                      <p className="mt-0.5 text-[7px] text-muted-foreground">
                        No current unusual-response signal
                      </p>
                    </div>

                    <ChevronRight className="h-3 w-3 text-muted-foreground" />
                  </button>
                ),
              )}

              {athletesWithoutReviewSignal.length ===
                0 && (
                <div className="rounded-lg border border-dashed border-green-100 bg-white px-3 py-4 text-center">
                  <p className="text-[8px] font-semibold">
                    All athletes currently have a review signal
                  </p>
                </div>
              )}
            </div>
          </div>
        </div>

        <p className="mt-2 text-[7px] text-muted-foreground">
          Priority reflects athlete-reported check-in patterns. It is a review aid, not a diagnosis or automatic coaching decision.
        </p>
      </section>

      <div className="grid grid-cols-[1fr_0.9fr_1.12fr] gap-3">
        <section className="rounded-2xl border border-border bg-white p-2.5 shadow-sm">
          <div className="flex items-start justify-between gap-2">
            <div>
              <h3 className="text-xs font-bold">
                1. How are athletes responding?
              </h3>

              <p className="mt-1 text-[8px] text-muted-foreground">
                Trends across the selected period
              </p>
            </div>

            <div className="flex rounded-lg bg-muted/40 p-0.5">
              <button
                type="button"
                onClick={() => setTrendMode("average")}
                className={[
                  "rounded-md px-2 py-1 text-[7px] font-semibold",
                  trendMode === "average"
                    ? "bg-white text-violet-700 shadow-sm"
                    : "text-muted-foreground",
                ].join(" ")}
              >
                Team Average

function getPerformanceFactorIcon(
  label: string,
) {
  const normalized =
    label.toLowerCase();

  if (
    normalized.includes("schlaf") ||
    normalized.includes("sleep")
  ) {
    return MoonStar;
  }

  if (
    normalized.includes("stress") ||
    normalized.includes("druck") ||
    normalized.includes("pressure")
  ) {
    return Brain;
  }

  if (
    normalized.includes("schule") ||
    normalized.includes("school") ||
    normalized.includes("arbeit") ||
    normalized.includes("work")
  ) {
    return Activity;
  }

  if (
    normalized.includes("wetter") ||
    normalized.includes("weather")
  ) {
    return Zap;
  }

  if (
    normalized.includes("training")
  ) {
    return Target;
  }

  if (
    normalized.includes("wettkampf") ||
    normalized.includes("competition")
  ) {
    return Activity;
  }

  return Activity;
}

function getPerformanceFactorIcon(
  label: string,
) {
  const normalized =
    label.toLowerCase();

  if (
    normalized.includes("schlaf") ||
    normalized.includes("sleep")
  ) {
    return MoonStar;
  }

  if (
    normalized.includes("stress") ||
    normalized.includes("druck") ||
    normalized.includes("pressure")
  ) {
    return Brain;
  }

  if (
    normalized.includes("schule") ||
    normalized.includes("school") ||
    normalized.includes("arbeit") ||
    normalized.includes("work")
  ) {
    return Activity;
  }

  if (
    normalized.includes("wetter") ||
    normalized.includes("weather")
  ) {
    return Zap;
  }

  if (
    normalized.includes("training")
  ) {
    return Target;
  }

  if (
    normalized.includes("wettkampf") ||
    normalized.includes("competition")
  ) {
    return Activity;
  }

  return Activity;
}

export const saveDailyCheckin = createServerFn({ method: "POST" })
  .middleware([requireSupabaseAuth])
  .inputValidator((input) => checkinSchema.parse(input))
  .handler(async ({ data, context }) => {
    const { supabase, userId } = context;
    const today = new Date().toISOString().slice(0, 10);

    // Build compact factual context summary for future Athletic IQ analyses.
    let context_summary: string | null = null;

    try {
      const ai = await callAi({
        model: "google/gemini-3.5-flash",
        messages: [
          {
            role: "system",
            content: buildDailyCheckinSummaryPrompt(DAILY_CHECKIN_CONTEXT_LIMIT),
          },
          {
            role: "user",
            content: JSON.stringify(data),
          },
        ],
      });

      const reply = ai?.choices?.[0]?.message?.content;

      if (typeof reply === "string" && reply.trim()) {
        context_summary = reply.trim().slice(0, 200);
      }
    } catch (error) {
      console.error("Daily Check-In context summary failed:", error);
    }

    const { data: row, error } = await supabase
      .from("daily_checkins")
      .upsert(
        {
          ...data,
          performance_results: data.performance_results as Json | null,
          user_id: userId,
          checkin_date: today,
          context_summary,
          updated_at: new Date().toISOString(),
        },
        { onConflict: "user_id,checkin_date" },
      )
      .select()
      .single();
    if (error) throw new Error(error.message);

    // Auto-create workout + analysis board for training days
    let boardId: string | null = null;
    if (!data.is_rest_day && data.discipline) {
      try {
        // Resolve athlete name for personal touch
        const { data: profile } = await supabase
          .from("profiles").select("display_name").eq("id", userId).maybeSingle();
        const name = profile?.display_name ?? "Athlet";

        const intensity: "low" | "moderate" | "high" =
          (data.rpe ?? 0) >= 8 ? "high" : (data.rpe ?? 0) >= 5 ? "moderate" : "low";
        const summary = `${data.discipline}-Training`;

        // Keep only one auto check-in workout per day
        await supabase.from("workouts")
          .delete().eq("user_id", userId).eq("workout_date", today)
          .eq("raw_input", `Daily check-in: ${data.discipline}`);

        const { data: workout } = await supabase
          .from("workouts")
          .insert({
            user_id: userId,
            workout_date: today,
            summary,
            duration_minutes: data.duration_min,
            intensity,
            raw_input: `Daily check-in: ${data.discipline}`,
          })
          .select()
          .single();

        if (workout) {
          /*
          ===============================================================
                     LOAD DAILY CHECK-IN HISTORY
          ===============================================================

          Prompt 2 should understand today's training in the context of
          the athlete's recent Daily Check-Ins.

          Rules:
          - Only load check-ins belonging to this athlete.
          - Only load days BEFORE today.
          - Ignore rows without a context_summary.
          - Load newest summaries first.
          - Never load more than DAILY_CHECKIN_CONTEXT_LIMIT.
          - If the athlete has fewer summaries, simply use those available.

          Examples:
          Check-in #1  -> 0 previous summaries
          Check-in #5  -> 4 previous summaries
          Check-in #31 -> 30 previous summaries
          ===============================================================
          */
          const { data: previousCheckins, error: historyError } =
            await supabase
              .from("daily_checkins")
              .select("checkin_date, context_summary")
              .eq("user_id", userId)
              .lt("checkin_date", today)
              .not("context_summary", "is", null)
              .order("checkin_date", { ascending: false })
              .limit(DAILY_CHECKIN_CONTEXT_LIMIT);

          if (historyError) {
            console.error(
              "Could not load previous Daily Check-In summaries:",
              historyError,
            );
          }

          // Always pass an array to Prompt 2.
          // If history loading fails or no previous logs exist, use [].
          const history = previousCheckins ?? [];

          /*
          TEMPORARY HISTORY DEBUG

          Prints exactly what Prompt 2 receives from previous Daily Check-Ins.
          Remove after historical-context testing is finished.
          */
          console.log("DAILY CHECK-IN HISTORY SENT TO PROMPT 2:", {
            count: history.length,
            history,
          });

          const analysis = await generateCheckinAnalysis(
            name,
            data,
            history,
          );
          const { data: b } = await supabase
            .from("analysis_boards")
            .insert({
              workout_id: workout.id,
              user_id: userId,
              positives: analysis.positives,
              negatives: analysis.negatives,
              improvements: analysis.improvements,
              is_ai_generated: true,
            })
            .select()
            .single();
          boardId = b?.id ?? null;
        }
      } catch (error) {
        console.error("generateCheckinAnalysis FAILED:", error);
        // non-fatal: check-in already saved
      }
    }
    return { checkin: row, boardId };
  });

async function generateCheckinAnalysis(
  name: string,
  c: z.infer<typeof checkinSchema>,

  // Previous Daily Check-In summaries used as historical evidence.
  // There can be 0 to DAILY_CHECKIN_CONTEXT_LIMIT entries.
  previousCheckins: Array<{
    checkin_date: string;
    context_summary: string | null;
  }>,
): Promise<AnalysisOutput> {
  try {
    console.log("CHECK-IN DATA SENT TO AI:", c);
    const ai = await callAi({
      model: "google/gemini-3.5-flash",
      messages: [
        {
          role: "system",
          content: buildDailyCheckinAnalysisPrompt(name, DAILY_CHECKIN_CONTEXT_LIMIT),
        },
        {
          role: "user",

          /*
          Prompt 2 receives TWO separate sources:

          1. today's full raw Daily Check-In
             -> best source for understanding today's session

          2. previous compact Daily Check-In summaries
             -> historical evidence for possible patterns/trends

          We keep them separate so Gemini knows which information
          describes TODAY and which information describes HISTORY.
          */
          content: JSON.stringify({
            today: c,
            previous_daily_checkins: previousCheckins,
          }),
        },
      ],
      tools: [
        {
          type: "function",
          function: {
            name: "post_checkin_analysis",
            parameters: {
              type: "object",
              properties: {
                positives: { type: "array", items: { type: "string" }, minItems: 1, maxItems: 2 },
                negatives: { type: "array", items: { type: "string" }, minItems: 1, maxItems: 2 },
                improvements: { type: "array", items: { type: "string" }, minItems: 1, maxItems: 2 },
              },
              required: ["positives", "negatives", "improvements"],
              additionalProperties: false,
            },
          },
        },
      ],
      tool_choice: { type: "function", function: { name: "post_checkin_analysis" } },
    });
    const call = ai?.choices?.[0]?.message?.tool_calls?.[0];
    if (call) return JSON.parse(call.function.arguments);
  } catch (error) {
    console.error("Check-in AI analysis failed:", error);
    // fallthrough to fallback
  }
  return {
    positives: [
      "Dein Check-In wurde gespeichert. Für eine verlässliche positive Bewertung liegen diesmal nicht genug eindeutige Informationen vor.",
    ],
    negatives: [
      "Aus den vorhandenen Angaben lässt sich kein belastbarer Verbesserungspunkt ableiten, ohne etwas zu vermuten.",
    ],
    improvements: [
      "Halte beim nächsten vergleichbaren Training besonders Trainingsart, RPE, Leistungsqualität und ein konkretes Ergebnis oder eine wichtige Beobachtung fest.",
    ],
  };
}

function round(value: number, decimals = 1) {
  const multiplier = 10 ** decimals;
  return Math.round(value * multiplier) / multiplier;
}

function calculatePercentage(
  count: number,
  total: number,
): number | null {
  if (total === 0) return null;
  return round((count / total) * 100);
}

function percentageResult(
  count: number,
  total: number,
): PercentageResult {
  return {
    count,
    total,
    percentage: calculatePercentage(count, total),
  };
}

function getTrainingSessions(
  checkins: DailyCheckinForAnalysis[],
) {
  return checkins.filter(

   QUESTION 2 — AVERAGE RPE
------------------------------------------------------- */

export function calculateAverageRpe(
  checkins: DailyCheckinForAnalysis[],
): number | null {
  const values = getTrainingSessions(checkins)
    .map((checkin) => checkin.rpe)
    .filter(
      (value): value is number =>
        typeof value === "number" &&
        value >= 1 &&
        value <= 10,
    );

  if (values.length === 0) return null;

  const total = values.reduce(
    (sum, value) => sum + value,
    0,
  );

  return round(total / values.length);
}

/**
 * DAILY CHECK-IN COMPARISONS
 *
 * Compares answers that become more useful when viewed together.
 *
 * Current comparison:
 *
 * Question 5 — Goal Achievement
 * x
 * Question 4 — Performance Quality
 */

function round(value: number, decimals = 1) {
  const multiplier = 10 ** decimals;
  return Math.round(value * multiplier) / multiplier;
}

export function calculateGoalPerformanceComparison(
  checkins: DailyCheckinForAnalysis[],
): GoalPerformanceComparison {
  /**
   * Excluded:
   *
   * Q5: "Weiß ich nicht"
   * Q4: "Kann ich nicht beurteilen"
   *
   * They cannot be meaningfully placed into this comparison.
   */
  const validSessions = checkins.filter(
    (checkin) =>
      checkin.is_rest_day !== true &&
      GOAL_ANSWERS.includes(
        checkin.session_goal_achieved as
          (typeof GOAL_ANSWERS)[number],
      ) &&
      PERFORMANCE_QUALITY_ANSWERS.includes(
        checkin.performance_quality as
          (typeof PERFORMANCE_QUALITY_ANSWERS)[number],
      ),
  );

  const totalSessions = validSessions.length;

  const cells = GOAL_ANSWERS.flatMap((goal) =>
    PERFORMANCE_QUALITY_ANSWERS.map(
      (performance) => {
        const count = validSessions.filter(
          (checkin) =>
            checkin.session_goal_achieved === goal &&
            checkin.performance_quality === performance,
        ).length;

        return {
          goal,
          performance,
          count,
          percentage:
            totalSessions === 0
              ? null
              : round(
                  (count / totalSessions) * 100,
                ),
        };
      },
    ),
  );

  return {
    totalSessions,
    cells,
  };
}
